import { useState, useRef } from "react";

function App() {
  const [file, setFile] = useState(null);
  const [page, setPage] = useState("home");

  const [error, setError] = useState("");

  const fileInputRef = useRef(null);
  const folderInputRef = useRef(null);

  const outlierCount = Number.isFinite(result?.outliers)
    ? result.outliers
    : null;

  const processFiles = async (selectedFiles) => {
    if (!selectedFiles || selectedFiles.length === 0) {
      return;
    }

    const filesArray = Array.from(selectedFiles);

    /*
     * Do not reject large datasets on the frontend with
     * an unnecessarily small limit.
     *
     * Backend controls the actual HTTP limit.
     */
    const totalSize = filesArray.reduce(
      (acc, f) => acc + f.size,
      0
    );

    const firstFile = filesArray[0];

    const isFolder =
      filesArray.length > 1 ||
      (
        firstFile.webkitRelativePath &&
        firstFile.webkitRelativePath !== ""
      );

    const displayName = isFolder
      ? (
          firstFile.webkitRelativePath
            ? firstFile.webkitRelativePath.split("/")[0]
            : `${filesArray.length} files`
        )
      : firstFile.name;

    setFile({
      name: displayName,
      size: totalSize,
      isFolder,
      count: filesArray.length,
    });

    setPage("dashboard");
    setLoading(true);
    setError("");
    setResult(null);

    const formData = new FormData();

    filesArray.forEach((selectedFile) => {
      formData.append("files", selectedFile);
    });

    try {
      const response = await fetch(
        "http://127.0.0.1:5000/api/upload",
        {
          method: "POST",
          body: formData,
        }
      ).catch(() => {
        throw new Error(
          "Cannot connect to server. Ensure Flask backend is running at http://127.0.0.1:5000"
        );
      });

      const contentType =
        response.headers.get("content-type") || "";

      if (response.status === 413) {
        throw new Error(
          "Dataset is larger than the server upload limit. Increase MAX_GB in backend/app.py or upload the dataset in smaller batches."
        );
      }

      if (
        contentType.includes("application/json")
      ) {
        const data = await response.json();

        if (!response.ok) {
          throw new Error(
            data.message ||
              "Upload processing failed."
          );
        }

        setResult(data);
      } else {
        const rawText =
          await response.text();

        console.error(
          "Non-JSON response:",
          rawText
        );

        throw new Error(
          `Server returned HTTP ${response.status}. Check Flask backend console.`
        );
      }
    } catch (err) {
      console.error(err);
      setError(
        err?.message ||
          "Upload processing failed."
      );
    } finally {
      setLoading(false);
    }
  };

    <div className="app">

      {/* FILE INPUT */}
      <input
        type="file"
        ref={fileInputRef}
        multiple
        accept="
          .csv,
          .xlsx,

          .parquet,
          .txt,
          .pdf,
          .ipynb,
          .zip,
          .png,
          .jpg,
          .jpeg,
          .webp,
          .gif,
          .bmp,
          .tif,
          .tiff
        "

          processFiles(e.target.files)

/* ========================================================
   HELPERS
======================================================== */

function formatPercent(value) {

  if (
    value === null ||
    value === undefined ||
    !Number.isFinite(
      Number(value)
    )
  ) {
    return "—";
  }

  return `${(
    Number(value) * 100
  ).toFixed(1)}%`;
}

def safe_int(value, default=0):
    try:
        return int(value)
    except Exception:
        return default


def safe_float(value, default=0.0):
    try:
        value = float(value)
        if np.isfinite(value):
            return value
        return default
    except Exception:
        return default


def format_bytes(size):
    size = float(size)

    units = ["B", "KB", "MB", "GB", "TB"]

    for unit in units:
        if size < 1024:
            return f"{size:.2f} {unit}"
        size /= 1024

    return f"{size:.2f} PB"


def clean_column_names(df):
    df = df.copy()

    new_columns = []

    for i, column in enumerate(df.columns):
        name = str(column).strip()

        if not name:
            name = f"column_{i + 1}"

        new_columns.append(name)

    # Make duplicate names unique.
    seen = {}

    final_columns = []

    for name in new_columns:
        if name not in seen:
            seen[name] = 0
            final_columns.append(name)
        else:
            seen[name] += 1
            final_columns.append(f"{name}_{seen[name]}")

    df.columns = final_columns

    return df


# ---------------------------------------------------------
# OUTLIER DETECTION
# ---------------------------------------------------------

def count_iqr_outliers(df):
    numeric = df.select_dtypes(include=[np.number])

    if numeric.empty:
        return 0

    try:
        q1 = numeric.quantile(0.25)
        q3 = numeric.quantile(0.75)

        iqr = q3 - q1

        lower = q1 - 1.5 * iqr
        upper = q3 + 1.5 * iqr

        mask = (numeric.lt(lower)) | (numeric.gt(upper))

        return int(mask.sum().sum())

    except Exception:
        return 0

def profile_csv_stream(file_storage):
    """
    Profiles CSV without loading the complete CSV into RAM.

    Uses chunks for large files.
    """

    file_storage.stream.seek(0)

    total_rows = 0
    total_missing = 0
    total_duplicates = 0
    total_outliers = 0

    column_names = []
    dtypes = {}

    numeric_values = {}

    try:
        for chunk in pd.read_csv(
            file_storage.stream,
            chunksize=CSV_CHUNK_SIZE,
            low_memory=True
        ):

            chunk = clean_column_names(chunk)

            if not column_names:
                column_names = list(chunk.columns.astype(str))

            total_rows += len(chunk)

            total_missing += int(chunk.isna().sum().sum())

            # Duplicate detection inside chunks.
            total_duplicates += int(chunk.duplicated().sum())

            # Keep small numeric samples for fast outlier estimation.
            numeric = chunk.select_dtypes(include=[np.number])

            for col in numeric.columns:

                if col not in numeric_values:
                    numeric_values[col] = []

                values = numeric[col].dropna()

                if len(values) > 500:
                    values = values.sample(
                        n=500,
                        random_state=42
                    )

                numeric_values[col].extend(
                    values.tolist()
                )

            for col in chunk.columns:
                if col not in dtypes:
                    dtypes[col] = str(chunk[col].dtype)

        # Approximate IQR outliers using sampled numeric values.
        for col, values in numeric_values.items():

            if not values:
                continue

            arr = np.asarray(values, dtype=float)

            if len(arr) < 10:
                continue

            q1 = np.percentile(arr, 25)
            q3 = np.percentile(arr, 75)

            iqr = q3 - q1

            if iqr == 0:
                continue

            lower = q1 - 1.5 * iqr
            upper = q3 + 1.5 * iqr

            total_outliers += int(
                ((arr < lower) | (arr > upper)).sum()
            )

        return {
            "rows": total_rows,
            "columns": len(column_names),
            "missing_values": total_missing,
            "duplicate_rows": total_duplicates,
            "outliers": total_outliers,
            "column_names": column_names,
            "dtypes": dtypes,
            "processing": "chunked",
        }

    except Exception as e:

        print(f"CSV profiling error: {e}")

        return None

def load_tabular_sample(file_storage, filename, max_rows=MAX_ML_ROWS):

    ext = os.path.splitext(filename.lower())[1]

    try:

        file_storage.stream.seek(0)

        if ext == ".csv":

            df = pd.read_csv(
                file_storage.stream,
                nrows=max_rows,
                low_memory=True
            )

        elif ext in [".xlsx", ".xls"]:

            # Excel readers generally need more memory than CSV.
            df = pd.read_excel(
                file_storage.stream,
                nrows=max_rows
            )

        elif ext == ".parquet":

            df = pd.read_parquet(
                file_storage.stream
            )

            if len(df) > max_rows:
                df = df.sample(
                    n=max_rows,
                    random_state=42
                )

        elif ext == ".json":

            df = pd.read_json(
                file_storage.stream
            )

            if len(df) > max_rows:
                df = df.sample(
                    n=max_rows,
                    random_state=42
                )

        else:
            return None

        return clean_column_names(df)

    except Exception as e:

        print(
            f"Could not load sample for {filename}: {e}"
        )

        return None

def profile_tabular_file(file_storage, filename):

    ext = os.path.splitext(filename.lower())[1]

    # CSV uses chunk processing.
    if ext == ".csv":

        return profile_csv_stream(file_storage)

    try:

        file_storage.stream.seek(0)

        if ext in [".xlsx", ".xls"]:

            df = pd.read_excel(
                file_storage.stream,
                nrows=MAX_DEMO_ROWS
            )

            # Get approximate/full row count where possible.
            sampled = clean_column_names(df)

            return build_dataframe_profile(
                sampled,
                processing="sampled_excel"
            )

        if ext == ".parquet":

            df = pd.read_parquet(
                file_storage.stream
            )

            return build_dataframe_profile(
                df,
                processing="parquet"
            )

        if ext == ".json":

            df = pd.read_json(
                file_storage.stream
            )

            return build_dataframe_profile(
                df,
                processing="json"
            )

    except Exception as e:

        print(
            f"Tabular parsing error for {filename}: {e}"
        )

    return None


def build_dataframe_profile(df, processing="normal"):

    if df is None:
        return None

    df = clean_column_names(df)

    rows, cols = df.shape

    missing = int(
        df.isna().sum().sum()
    )

    duplicates = int(
        df.duplicated().sum()
    )

    outliers = count_iqr_outliers(df)

    dtypes = {
        str(col): str(dtype)
        for col, dtype in df.dtypes.items()
    }

    return {
        "rows": int(rows),
        "columns": int(cols),
        "missing_values": missing,
        "duplicate_rows": duplicates,
        "outliers": outliers,
        "column_names": [
            str(c) for c in df.columns
        ],
        "dtypes": dtypes,
        "processing": processing,
    }

def parse_pdf(file_storage):

    result = {
        "rows": 0,
        "columns": 1,
        "missing_values": 0,
        "duplicate_rows": 0,
        "outliers": 0,
        "column_names": [],
        "status": "success",
        "reason": "",
        "pdf_pages": 0,
        "pdf_words": 0,
    }

    if pypdf is None:

        result["status"] = "skipped"
        result["reason"] = (
            "pypdf is not installed. "
            "Run: pip install pypdf"
        )

        return result

    try:

        file_storage.stream.seek(0)

        reader = pypdf.PdfReader(
            file_storage.stream,
            strict=False
        )

        if reader.is_encrypted:

            try:
                reader.decrypt("")
            except Exception:
                pass

            if reader.is_encrypted:

                result["status"] = "skipped"
                result["reason"] = (
                    "PDF is password-protected. "
                    "Upload an unlocked PDF."
                )

                return result

        pages = len(reader.pages)

        word_count = 0

        for page in reader.pages:

            try:
                text = page.extract_text() or ""
                word_count += len(text.split())
            except Exception:
                continue

        result.update({
            "rows": pages,
            "pdf_pages": pages,
            "pdf_words": word_count,
            "column_names": [
                f"PDF pages: {pages}",
                f"Extracted words: {word_count}"
            ],
        })

        # Scanned/image-only PDF.
        if word_count == 0:

            result["reason"] = (
                "PDF contains no extractable text. "
                "It may be scanned/image-based."
            )

        return result

    except Exception as e:

        error_text = str(e).lower()

        if (
            "password" in error_text
            or "encrypted" in error_text
        ):

            result["status"] = "skipped"
            result["reason"] = (
                "PDF is password-protected."
            )

        else:

            result["status"] = "skipped"
            result["reason"] = (
                "PDF is damaged, truncated, invalid, "
                "or not a real PDF."
            )

        print(
            f"PDF skipped: {file_storage.filename} -> {e}"
        )

        return result

def parse_text_or_code(file_storage, filename):

    try:

        file_storage.stream.seek(0)

        raw = file_storage.stream.read(
            5 * 1024 * 1024
        )

        text = raw.decode(
            "utf-8",
            errors="ignore"
        )

        if filename.lower().endswith(".ipynb"):

            if nbformat is None:

                return {
                    "rows": 0,
                    "columns": 0,
                    "missing_values": 0,
                    "duplicate_rows": 0,
                    "outliers": 0,
                    "column_names": [],
                    "status": "skipped",
                    "reason": "nbformat is not installed."
                }

            try:

                nb = nbformat.reads(
                    text,
                    as_version=4
                )

                code_cells = sum(
                    1
                    for c in nb.cells
                    if c.cell_type == "code"
                )

                markdown_cells = sum(
                    1
                    for c in nb.cells
                    if c.cell_type == "markdown"
                )

                return {
                    "rows": len(nb.cells),
                    "columns": 2,
                    "missing_values": 0,
                    "duplicate_rows": 0,
                    "outliers": 0,
                    "column_names": [
                        f"Code cells: {code_cells}",
                        f"Markdown cells: {markdown_cells}",
                    ],
                    "status": "success",
                }

            except Exception:

                return {
                    "rows": len(text.splitlines()),
                    "columns": 1,
                    "missing_values": 0,
                    "duplicate_rows": 0,
                    "outliers": 0,
                    "column_names": [
                        "Notebook text"
                    ],
                    "status": "success",
                }

        lines = text.splitlines()

        return {
            "rows": len(lines),
            "columns": 1,
            "missing_values": 0,
            "duplicate_rows": 0,
            "outliers": 0,
            "column_names": [
                f"Text lines: {len(lines)}",
                f"Words: {len(text.split())}",
            ],
            "status": "success",
        }

    except Exception as e:

        return {
            "rows": 0,
            "columns": 0,
            "missing_values": 0,
            "duplicate_rows": 0,
            "outliers": 0,
            "column_names": [],
            "status": "skipped",
            "reason": str(e),
        }

def parse_zip(file_storage):

    result = {
        "rows": 0,
        "columns": 0,
        "missing_values": 0,
        "duplicate_rows": 0,
        "outliers": 0,
        "column_names": [],
        "status": "success",
        "zip_files": 0,
        "image_files": 0,
        "tabular_files": 0,
    }

    try:

        file_storage.stream.seek(0)

        with zipfile.ZipFile(
            file_storage.stream
        ) as z:

            names = z.namelist()

            result["zip_files"] = len(names)

            for name in names:

                lower = name.lower()

                if lower.endswith(
                    (
                        ".csv",
                        ".xlsx",
                        ".xls",
                        ".json",
                        ".parquet",
                    )
                ):
                    result["tabular_files"] += 1

                elif any(
                    lower.endswith(ext)
                    for ext in IMAGE_EXTENSIONS
                ):
                    result["image_files"] += 1

            result["column_names"] = [
                f"ZIP files: {len(names)}",
                f"Tabular files: {result['tabular_files']}",
                f"Image files: {result['image_files']}",
            ]

        return result

    except Exception as e:

        return {
            **result,
            "status": "skipped",
            "reason": f"Invalid ZIP file: {e}",
        }

def clean_dataframe(df):

    if df is None or df.empty:

        return df, {
            "rows_before": 0,
            "rows_after": 0,
            "missing_before": 0,
            "missing_after": 0,
            "duplicates_before": 0,
            "duplicates_after": 0,
            "outliers_before": 0,
            "outliers_after": 0,
        }

    df = clean_column_names(df.copy())

    rows_before = len(df)

    missing_before = int(
        df.isna().sum().sum()
    )

    duplicates_before = int(
        df.duplicated().sum()
    )

    outliers_before = count_iqr_outliers(df)

    # -----------------------------------------------------
    # REMOVE DUPLICATES
    # -----------------------------------------------------

    df = df.drop_duplicates()

    # -----------------------------------------------------
    # MISSING VALUES
    # -----------------------------------------------------

    numeric_columns = df.select_dtypes(
        include=[np.number]
    ).columns

    categorical_columns = df.select_dtypes(
        exclude=[np.number]
    ).columns

    for column in numeric_columns:

        if df[column].isna().any():

            median = df[column].median()

            if pd.isna(median):
                median = 0

            df[column] = df[column].fillna(
                median
            )

    for column in categorical_columns:

        if df[column].isna().any():

            mode = df[column].mode()

            if not mode.empty:

                value = mode.iloc[0]

            else:

                value = "Unknown"

            df[column] = df[column].fillna(
                value
            )

    # -----------------------------------------------------
    # OUTLIER CLIPPING
    # -----------------------------------------------------

    for column in numeric_columns:

        try:

            q1 = df[column].quantile(0.25)
            q3 = df[column].quantile(0.75)

            iqr = q3 - q1

            if pd.isna(iqr) or iqr == 0:
                continue

            lower = q1 - 1.5 * iqr
            upper = q3 + 1.5 * iqr

            df[column] = df[column].clip(
                lower,
                upper
            )

        except Exception:
            continue

    missing_after = int(
        df.isna().sum().sum()
    )

    duplicates_after = int(
        df.duplicated().sum()
    )

    outliers_after = count_iqr_outliers(df)

    return df, {
        "rows_before": rows_before,
        "rows_after": len(df),
        "missing_before": missing_before,
        "missing_after": missing_after,
        "duplicates_before": duplicates_before,
        "duplicates_after": duplicates_after,
        "outliers_before": outliers_before,
        "outliers_after": outliers_after,
    }

# TARGET DETECTION
# ---------------------------------------------------------

def infer_target_column(df):

    if df is None or df.empty:
        return None

    columns = list(df.columns)

    # Prefer common target names.
    target_words = [
        "target",
        "label",
        "class",
        "output",
        "result",
        "y",
        "category",
        "diagnosis",
        "churn",
        "fraud",
    ]

    for word in target_words:

        for column in columns:

            if str(column).lower().strip() == word:

                return column

    # Prefer low-cardinality categorical column.
    candidates = []

    for column in columns:

        unique_count = df[column].nunique(
            dropna=True
        )

        if (
            unique_count >= 2
            and unique_count <= MAX_CLASSES
        ):

            candidates.append(
                (column, unique_count)
            )

    if candidates:

        return candidates[-1][0]

    # Last column fallback.
    if columns:
        return columns[-1]

    return None

# FEATURE SELECTION + MODEL EVALUATION
# ---------------------------------------------------------

def build_pipeline(
    pipeline_name,
    numeric_columns,
    categorical_columns,
    selector
):

    transformers = []

    if numeric_columns:

        transformers.append(
            (
                "numeric",
                Pipeline([
                    (
                        "imputer",
                        SimpleImputer(
                            strategy="median"
                        )
                    ),
                    (
                        "scaler",
                        StandardScaler()
                    ),
                ]),
                numeric_columns,
            )
        )

    if categorical_columns:

        transformers.append(
            (
                "categorical",
                Pipeline([
                    (
                        "imputer",
                        SimpleImputer(
                            strategy="most_frequent"
                        )
                    ),
                    (
                        "onehot",
                        OneHotEncoder(
                            handle_unknown="ignore"
                        ),
                    ),
                ]),
                categorical_columns,
            )
        )

    preprocessor = ColumnTransformer(
        transformers=transformers
    )

    if pipeline_name == "Pipeline A":

        model = LogisticRegression(
            max_iter=500,
            n_jobs=None
        )

    elif pipeline_name == "Pipeline B":

        model = RandomForestClassifier(
            n_estimators=120,
            random_state=42,
            n_jobs=-1,
            max_depth=14
        )

    else:

        model = ExtraTreesClassifier(
            n_estimators=120,
            random_state=42,
            n_jobs=-1,
            max_depth=14
        )

    steps = [
        ("preprocessing", preprocessor)
    ]

    if selector is not None:

        steps.append(
            ("feature_selection", selector)
        )

    steps.append(
        ("model", model)
    )

    return Pipeline(steps)


def evaluate_tabular_dataset(df):

    empty_result = {
        "status": "unavailable",
        "reason": "",
        "target_column": None,
        "selected_features": [],
        "pipelines": [],
        "best_pipeline": None,
        "model": None,
        "training_seconds": 0,
    }

    if not SKLEARN_AVAILABLE:

        empty_result["reason"] = (
            "scikit-learn is not installed. "
            "Run: pip install scikit-learn"
        )

        return empty_result

    if df is None or df.empty:

        empty_result["reason"] = (
            "Dataset is empty."
        )

        return empty_result

    # -----------------------------------------------------
    # LIMIT NUMBER OF COLUMNS
    # -----------------------------------------------------

    if df.shape[1] > MAX_FEATURE_COLUMNS:

        # Keep columns with useful variance.
        numeric_cols = list(
            df.select_dtypes(
                include=[np.number]
            ).columns
        )

        other_cols = [
            c for c in df.columns
            if c not in numeric_cols
        ]

        selected_columns = (
            numeric_cols[:MAX_FEATURE_COLUMNS]
            + other_cols[:MAX_FEATURE_COLUMNS]
        )

        selected_columns = list(
            dict.fromkeys(selected_columns)
        )[:MAX_FEATURE_COLUMNS]

        df = df[selected_columns]

    target = infer_target_column(df)

    if target is None:

        empty_result["reason"] = (
            "Could not automatically detect target column."
        )

        return empty_result

    if target not in df.columns:

        empty_result["reason"] = (
            "Detected target column is unavailable."
        )

        return empty_result

    # Remove rows without target.
    df = df.dropna(
        subset=[target]
    )

    if len(df) < 30:

        empty_result["reason"] = (
            "At least 30 labelled rows are required "
            "for model comparison."
        )

        return empty_result

    # -----------------------------------------------------
    # TARGET
    # -----------------------------------------------------

    y = df[target].astype(str)

    if y.nunique() < 2:

        empty_result["reason"] = (
            "Target contains only one class."
        )

        return empty_result

    if y.nunique() > MAX_CLASSES:

        empty_result["reason"] = (
            "Target has too many classes for "
            "the automatic classification comparison."
        )

        return empty_result

    # -----------------------------------------------------
    # REPRESENTATIVE SAMPLE
    # -----------------------------------------------------

    if len(df) > MAX_ML_ROWS:

        # Stratified sample if possible.
        try:

            _, df = train_test_split(
                df,
                test_size=MAX_ML_ROWS,
                random_state=42,
                stratify=y
            )

        except Exception:

            df = df.sample(
                n=MAX_ML_ROWS,
                random_state=42
            )

        y = df[target].astype(str)

    X = df.drop(
        columns=[target]
    )

    # Remove columns with no information.
    constant_columns = [
        col
        for col in X.columns
        if X[col].nunique(dropna=False) <= 1
    ]

    if constant_columns:

        X = X.drop(
            columns=constant_columns
        )

    if X.shape[1] == 0:

        empty_result["reason"] = (
            "No usable feature columns remain."
        )

        return empty_result

    numeric_columns = list(
        X.select_dtypes(
            include=[np.number]
        ).columns
    )

    categorical_columns = list(
        X.select_dtypes(
            exclude=[np.number]
        ).columns
    )

    # Encode target.
    label_encoder = LabelEncoder()

    y_encoded = label_encoder.fit_transform(y)

    # Train/test split.
    try:

        X_train, X_test, y_train, y_test = train_test_split(
            X,
            y_encoded,
            test_size=0.20,
            random_state=42,
            stratify=y_encoded
        )

    except Exception:

        X_train, X_test, y_train, y_test = train_test_split(
            X,
            y_encoded,
            test_size=0.20,
            random_state=42
        )

    # -----------------------------------------------------
    # FEATURE SELECTION
    # -----------------------------------------------------

    k = min(
        15,
        max(1, X.shape[1])
    )

    pipelines = []

    definitions = [
        (
            "Pipeline A",
            "Imputation + Scaling + Mutual Information",
            SelectKBest(
                score_func=mutual_info_classif,
                k=k
            ),
        ),
        (
            "Pipeline B",
            "Imputation + Scaling + Tree Feature Selection",
            None,
        ),
        (
            "Pipeline C",
            "Imputation + Scaling + Extra Trees",
            None,
        ),
    ]

    for pipeline_name, description, selector in definitions:

        start = time.perf_counter()

        try:

            pipeline = build_pipeline(
                pipeline_name,
                numeric_columns,
                categorical_columns,
                selector
            )

            pipeline.fit(
                X_train,
                y_train
            )

            predictions = pipeline.predict(
                X_test
            )

            accuracy = accuracy_score(
                y_test,
                predictions
            )

            f1 = f1_score(
                y_test,
                predictions,
                average="weighted",
                zero_division=0
            )

            balanced = balanced_accuracy_score(
                y_test,
                predictions
            )

            elapsed = (
                time.perf_counter()
                - start
            )

            # Extract selected feature information where possible.
            selected_features = []

            try:

                preprocessing = pipeline.named_steps[
                    "preprocessing"
                ]

                transformed_names = (
                    preprocessing
                    .get_feature_names_out()
                )

                if (
                    "feature_selection"
                    in pipeline.named_steps
                ):

                    selector_object = pipeline.named_steps[
                        "feature_selection"
                    ]

                    mask = selector_object.get_support()

                    selected_features = [
                        str(name)
                        for name, keep
                        in zip(
                            transformed_names,
                            mask
                        )
                        if keep
                    ]

                else:

                    # For tree pipelines, show top original
                    # numerical/categorical transformed features.
                    model = pipeline.named_steps[
                        "model"
                    ]

                    if hasattr(
                        model,
                        "feature_importances_"
                    ):

                        importances = (
                            model.feature_importances_
                        )

                        pairs = list(
                            zip(
                                transformed_names,
                                importances
                            )
                        )

                        pairs.sort(
                            key=lambda x: x[1],
                            reverse=True
                        )

                        selected_features = [
                            str(name)
                            for name, _ in pairs[:k]
                        ]

            except Exception as feature_error:

                print(
                    "Feature extraction warning:",
                    feature_error
                )

            pipelines.append({
                "name": pipeline_name,
                "description": description,
                "accuracy": round(
                    float(accuracy),
                    5
                ),
                "f1_weighted": round(
                    float(f1),
                    5
                ),
                "balanced_accuracy": round(
                    float(balanced),
                    5
                ),
                "training_seconds": round(
                    float(elapsed),
                    3
                ),
                "selected_feature_count": len(
                    selected_features
                ),
                "selected_features": selected_features[
                    :30
                ],
            })

        except Exception as pipeline_error:

            print(
                f"{pipeline_name} failed:",
                pipeline_error
            )

            elapsed = (
                time.perf_counter()
                - start
            )

            pipelines.append({
                "name": pipeline_name,
                "description": description,
                "accuracy": None,
                "f1_weighted": None,
                "balanced_accuracy": None,
                "training_seconds": round(
                    float(elapsed),
                    3
                ),
                "selected_feature_count": 0,
                "selected_features": [],
                "error": str(pipeline_error),
            })

    valid = [
        p
        for p in pipelines
        if p.get("accuracy") is not None
    ]

    if not valid:

        empty_result["reason"] = (
            "All automatic model pipelines failed."
        )

        empty_result["pipelines"] = pipelines

        return empty_result

    best = max(
        valid,
        key=lambda item: (
            item["balanced_accuracy"],
            item["accuracy"],
            item["f1_weighted"],
        )
    )

    return {
        "status": "complete",
        "target_column": str(target),
        "sample_rows": len(df),
        "original_rows": int(len(df)),
        "feature_count_before": int(X.shape[1]),
        "feature_count_after": int(
            best["selected_feature_count"]
        ),
        "selected_features": best[
            "selected_features"
        ],
        "pipelines": pipelines,
        "best_pipeline": best["name"],
        "best_score": best["accuracy"],
        "model": best["name"],
        "training_seconds": best[
            "training_seconds"
        ],
        "metric": "accuracy",
    }

# BEFORE / AFTER MODEL COMPARISON
# ---------------------------------------------------------

def run_before_after(df):

    if df is None or df.empty:

        return {
            "status": "unavailable",
            "before": None,
            "after": None,
            "change": None,
        }

    # Use a representative sample for expensive comparison.
    working = df.copy()

    if len(working) > MAX_DEMO_ROWS:

        working = working.sample(
            n=MAX_DEMO_ROWS,
            random_state=42
        )

    target = infer_target_column(
        working
    )

    if target is None:

        return {
            "status": "unavailable",
            "reason": "Target column could not be detected.",
        }

    try:

        before = evaluate_tabular_dataset(
            working
        )

        cleaned, cleaning = clean_dataframe(
            working
        )

        after = evaluate_tabular_dataset(
            cleaned
        )

        before_score = before.get(
            "best_score"
        )

        after_score = after.get(
            "best_score"
        )

        change = None

        if (
            before_score is not None
            and after_score is not None
        ):

            change = round(
                after_score - before_score,
                5
            )

        return {
            "status": "complete",
            "target_column": target,
            "cleaning": cleaning,
            "before": before,
            "after": after,
            "change": change,
        }

    except Exception as e:

        print(
            "Before/after evaluation error:",
            e
        )

        return {
            "status": "unavailable",
            "reason": str(e),
        }

def recommend_model(
    dataset_type,
    profile,
    evaluation
):

    if dataset_type == "tabular":

        best = evaluation.get(
            "best_pipeline"
        )

        if best:

            return {
                "model": best,
                "reason": (
                    "This pipeline achieved the highest "
                    "validation performance among the "
                    "automatically tested pipelines."
                ),
            }

        return {
            "model": "Random Forest",
            "reason": (
                "Suitable general-purpose model for "
                "mixed tabular data."
            ),
        }

    if dataset_type == "image":

        return {
            "model": "MobileNetV3 / Transfer Learning",
            "reason": (
                "Recommended for labelled image datasets "
                "because transfer learning can provide "
                "good accuracy with lower training cost "
                "than training a CNN from scratch."
            ),
        }

    if dataset_type == "pdf":

        return {
            "model": "Text classification after OCR/extraction",
            "reason": (
                "PDF files must first be converted into "
                "structured text/features."
            ),
        }

    if dataset_type == "text":

        return {
            "model": "TF-IDF + Logistic Regression",
            "reason": (
                "Fast baseline for text classification."
            ),
        }

    if dataset_type == "notebook":

        return {
            "model": "Dataset-dependent",
            "reason": (
                "Notebook files are source documents; "
                "the underlying dataset must be extracted "
                "before ML training."
            ),
        }

    if dataset_type == "zip":

        return {
            "model": "Depends on extracted dataset",
            "reason": (
                "ZIP is an archive. DataOpti first identifies "
                "the contents and then chooses a model."
            ),
        }

    return {
        "model": "Dataset-dependent",
        "reason": (
            "DataOpti needs structured data before "
            "automatic model comparison."
        ),
    }


# ---------------------------------------------------------
# DATASET TYPE
# ---------------------------------------------------------

def detect_type(filename):

    ext = os.path.splitext(
        filename.lower()
    )[1]

    if ext in [
        ".csv",
        ".xlsx",
        ".xls",
        ".json",
        ".parquet",
    ]:
        return "tabular"

    if ext == ".pdf":
        return "pdf"

    if ext in IMAGE_EXTENSIONS:
        return "image"

    if ext == ".zip":
        return "zip"

    if ext == ".ipynb":
        return "notebook"

    if ext in [
        ".txt",
        ".py",
        ".md",
        ".log",
    ]:
        return "text"

    return "unknown"

        # PROCESS EACH UPLOAD
        # -------------------------------------------------

        for file in files:

            filename = (
                file.filename or "unknown"
            )

            lower_name = filename.lower()

            if not filename:
                continue

            dataset_type = detect_type(
                filename
            )

            profile = None
            df_for_ml = None

            try:

                # -----------------------------------------
                # TABULAR
                # -----------------------------------------

                if dataset_type == "tabular":

                    profile = profile_tabular_file(
                        file,

                        )

                            profile.get(
                                "duplicate_rows"
                            )
                        )

                            profile.get(
                                "outliers"
                            )
                        )

                        all_columns.extend(
                            profile.get(
                                "column_names",
                                []
                            )
                        )

                    # Run full workflow only for the
                    # primary tabular dataset.
                    if (
                        df_for_ml is not None
                        and primary_evaluation is None
                    ):

                        primary_type = "tabular"
                        primary_profile = profile

                        cleaned_df, cleaning = (
                            clean_dataframe(
                                df_for_ml
                            )

                        )

                # IMAGE
